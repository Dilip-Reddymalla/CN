#include <iostream>
using namespace std;

int main() {
    string data, div, recv;

    cout << "Enter data bits: ";
    cin >> data;

    cout << "Enter divisor: ";
    cin >> div;

    int n = div.length();
    string temp = data + string(n - 1, '0');

    for (int i = 0; i < data.length(); i++) {
        if (temp[i] == '1') {
            for (int j = 0; j < n; j++)
                temp[i+j] = (temp[i+j] == div[j]) ? '0' : '1';
        }
    }

    string crc = temp.substr(data.length());
    string code = data + crc;

    cout << "CRC: " << crc << endl;
    cout << "Transmitted Data: " << code << endl;

    cout << "Enter received data: ";
    cin >> recv;

    for (int i = 0; i <= recv.length() - n; i++) {
        if (recv[i] == '1') {
            for (int j = 0; j < n; j++)
                recv[i+j] = (recv[i+j] == div[j]) ? '0' : '1';
        }
    }

    string rem = recv.substr(recv.length() - n + 1);

    if (rem.find('1') != string::npos)
        cout << "Error detected";
    else
        cout << "No error detected";

    return 0;
}
